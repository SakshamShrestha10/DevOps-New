name: Build and Deploy to AWS EC2 (Production)

on:
  push:
    tags:
      - '*'

jobs:
  build-deploy:
    runs-on: ubuntu-latest

    steps:
    # Checkout the repository
    - name: Checkout repository
      uses: actions/checkout@v3

    - name: Configure AWS credentials
      uses: aws-actions/configure-aws-credentials@v2
      with:
        aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
        aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
        aws-region: ${{ secrets.AWS_REGION }}

    # Log in to ECR
    - name: Log in to ECR
      run: |
        aws ecr get-login-password --region ${{ secrets.AWS_REGION }} | docker login --username AWS --password-stdin ${{ secrets.ECR_REPOSITORY_URI }}

    # Build and Push the Docker image
    - name: Build and Push Docker Image
      run: |
        TAG_NAME=${{ github.ref_name }}
        docker build -t devops-webapp -f Dockerfile-prod .
        docker tag devops-webapp:latest ${{ secrets.ECR_REPOSITORY_URI }}/devops-webapp:$TAG_NAME
        docker push ${{ secrets.ECR_REPOSITORY_URI }}/devops-webapp:$TAG_NAME
